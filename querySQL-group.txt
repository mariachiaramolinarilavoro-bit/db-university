-- 1. Contare quanti iscritti ci sono stati ogni anno

SELECT YEAR(enrolment_date) AS anno, COUNT(*) AS num_iscritti
FROM students
GROUP BY anno;

-- 2. Contare gli insegnanti che hanno l'ufficio nello stesso edificio

SELECT office_address, COUNT(*) AS num_insegnanti
FROM teachers
GROUP BY office_address;

-- 3. Calcolare la media dei voti di ogni appello d'esame

SELECT exams.id AS appello, courses.name AS corso, AVG(exam_student.vote) AS media_voti
FROM exams
JOIN courses ON exams.course_id = courses.id
JOIN exam_student ON exams.id = exam_student.exam_id
GROUP BY exams.id, courses.name;

-- 4. Contare quanti corsi di laurea ci sono per ogni dipartimento

SELECT departments.name AS dipartimento, COUNT(degrees.id) AS num_corsi_laurea
FROM departments
JOIN degrees ON departments.id = degrees.department_id
GROUP BY departments.id;
